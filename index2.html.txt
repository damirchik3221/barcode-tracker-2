    <script src="https://unpkg.com/html5-qrcode"></script>
    <script>
        let db = JSON.parse(localStorage.getItem('products') || '[]');
        let htmlScanner;

        function saveDb() { localStorage.setItem('products', JSON.stringify(db)); }

        function renderList() {
            const today = new Date(); today.setHours(0, 0, 0, 0);
            const warningDate = new Date(today); warningDate.setDate(today.getDate() + 3);
            const list = document.getElementById('product-list');
            list.innerHTML = '';

            if (db.length === 0) {
                list.innerHTML = '<p style="text-align:center;color:#999;">Нет продуктов. Добавьте первый!</p>';
                return;
            }

            db.sort((a, b) => new Date(a.exp) - new Date(b.exp));

            db.forEach((p, index) => {
                const expDate = new Date(p.exp);
                let statusClass, statusText;
                if (expDate < today) { statusClass = 'expired'; statusText = 'ПРОСРОЧЕН'; }
                else if (expDate <= warningDate) { statusClass = 'warning'; statusText = 'Истекает'; }
                else { statusClass = 'ok'; statusText = 'Нормально'; }

                const card = document.createElement('div');
                card.className = 'product-card';
                card.innerHTML = `
                    <div class="info">
                        <h3>${p.name}</h3>
                        <p>Штрих-код: ${p.barcode}</p>
                        <p>Годен до: <b>${p.exp}</b></p>
                    </div>
                    <div style="display: flex; flex-direction: column; align-items: flex-end; gap: 5px;">
                        <span class="status ${statusClass}">${statusText}</span>
                        <button class="delete-btn" onclick="deleteProduct(${index})">🗑️</button>
                    </div>
                `;
                list.appendChild(card);
            });
        }

        async function lookupProduct(barcode) {
            try {
                const response = await fetch(`https://world.openfoodfacts.org/api/v0/product/${barcode}.json`);
                const data = await response.json();
                if (data.status === 1 && data.product) {
                    return data.product.product_name || data.product.generic_name || '';
                }
            } catch (e) {
                console.log('Ошибка поиска:', e);
            }
            return '';
        }

        async function processBarcode(barcode) {
            const existing = db.find(p => p.barcode === barcode);
            if (existing) {
                const expDate = new Date(existing.exp);
                const today = new Date(); today.setHours(0, 0, 0, 0);
                const warningDate = new Date(today); warningDate.setDate(today.getDate() + 3);
                
                let statusText;
                if (expDate < today) statusText = 'ПРОСРОЧЕН!';
                else if (expDate <= warningDate) statusText = 'Истекает скоро';
                else statusText = 'Нормально';
                
                alert(`Продукт: ${existing.name}\nГоден до: ${existing.exp}\nСтатус: ${statusText}`);
                return;
            }

            // Открываем форму
            document.getElementById('inp-barcode').value = barcode;
            document.getElementById('inp-name').value = '';
            document.getElementById('add-title').textContent = 'Новый продукт';
            document.getElementById('add-modal').style.display = 'flex';

            // Ищем название
            document.getElementById('loading-indicator').style.display = 'block';
            document.getElementById('found-indicator').style.display = 'none';
            
            const productName = await lookupProduct(barcode);
            
            document.getElementById('loading-indicator').style.display = 'none';
            
            if (productName) {
                document.getElementById('inp-name').value = productName;
                document.getElementById('found-indicator').textContent = '✅ Найдено: ' + productName;
                document.getElementById('found-indicator').style.display = 'block';
            } else {
                document.getElementById('found-indicator').textContent = '❌ Название не найдено в базе';
                document.getElementById('found-indicator').style.display = 'block';
            }
        }

        function addProduct() {
            const name = document.getElementById('inp-name').value.trim();
            const barcode = document.getElementById('inp-barcode').value.trim();
            const exp = document.getElementById('inp-date').value;

            if (!name || !barcode || !exp) {
                alert('Заполните все поля!');
                return;
            }

            db.push({ name, barcode, exp });
            saveDb();
            renderList();
            closeAddModal();
        }

        function deleteProduct(index) {
            if (confirm('Удалить этот продукт?')) {
                db.splice(index, 1);
                saveDb();
                renderList();
            }
        }

        function openManualAdd() {
            document.getElementById('inp-name').value = '';
            document.getElementById('inp-barcode').value = '';
            document.getElementById('add-title').textContent = 'Добавить продукт';
            document.getElementById('loading-indicator').style.display = 'none';
            document.getElementById('found-indicator').style.display = 'none';
            document.getElementById('add-modal').style.display = 'flex';
        }

        function closeAddModal() {
            document.getElementById('add-modal').style.display = 'none';
            document.getElementById('loading-indicator').style.display = 'none';
            document.getElementById('found-indicator').style.display = 'none';
        }

        function startScan() {
            document.getElementById('scanner-modal').style.display = 'flex';
            htmlScanner = new Html5Qrcode("scanner-container");
            htmlScanner.start(
                { facingMode: "environment" },
                { fps: 10, qrbox: 250 },
                (decodedText) => {
                    stopScan();
                    processBarcode(decodedText);
                },
                (error) => {}
            ).catch(err => {
                alert('Ошибка камеры: ' + err.message);
                closeScannerModal();
            });
        }

        function stopScan() {
            if (htmlScanner) {
                htmlScanner.stop().then(() => closeScannerModal()).catch(() => closeScannerModal());
            } else {
                closeScannerModal();
            }
        }

        function closeScannerModal() {
            document.getElementById('scanner-modal').style.display = 'none';
        }

        document.getElementById('inp-date').value = new Date(Date.now() + 7*24*60*60*1000).toISOString().split('T')[0];

        renderList();
    </script>
